Hướng dẫn & Code Google Apps Script

Đồng bộ JSON giữa máy tính và điện thoại qua Google Sheets

  1. Mở Google Drive và tạo một file Google Sheets mới.
  2. Chọn Tiện ích mở rộng (Extensions) → Apps Script.
  3. Xóa code mặc định, dán đoạn code dưới đây và bấm Lưu.
  4. Chọn Triển khai (Deploy) → Tùy chọn triển khai mới (New deployment) → chọn Ứng dụng web (Web app): đặt Execute as: Me và Who has access: Anyone.
  5. Sao chép Web App URL, dán vào tab Tiến độ & Sync và bấm Lưu cấu hình.
CODE.GS 
const SECRET_TOKEN = "";

function jsonResponse(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function isAuthorized(token) {
  if (!SECRET_TOKEN) return true;
  return token === SECRET_TOKEN;
}

function doGet(e) {
  try {
    const token = (e && e.parameter && e.parameter.token) ? String(e.parameter.token) : "";
    if (!isAuthorized(token)) {
      return jsonResponse({ ok: false, error: "Invalid token" });
    }
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
    const raw = sheet.getRange("A1").getValue();
    return jsonResponse({ ok: true, payload: raw ? JSON.parse(raw) : null });
  } catch (err) {
    return jsonResponse({ ok: false, error: err.message });
  }
}

function doPost(e) {
  try {
    const body = JSON.parse(e.postData.contents || "{}");
    if (!isAuthorized(body.token || "")) {
      return jsonResponse({ ok: false, error: "Invalid token" });
    }
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
    sheet.getRange("A1").setValue(JSON.stringify(body.payload || {}));
    sheet.getRange("B1").setValue("Updated: " + new Date().toISOString());
    return jsonResponse({ ok: true, syncedAt: new Date().toISOString() });
  } catch (err) {
    return jsonResponse({ ok: false, error: err.message });
  }
}
